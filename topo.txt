APP
├─ README.md                          # Hướng dẫn/ghi chú dự án
├─ topo.txt                           # Sơ đồ dự án 
├─ data/                              # Bộ dữ liệu đầu vào (checklist)
│  ├─ canonical_ubuntu_24.04_lts.csv  # Checklist dạng CSV
│  ├─ canonical_ubuntu_24.04_lts.json # Checklist dạng JSON
│  └─ U_CAN_Ubuntu_24-04...xccdf.xml  # Checklist STIG/XCCDF dạng XML
├─ logs/                              # Thư mục chứa log theo từng phiên chạy (mặc định)
├─ pyproject.toml                     # Cấu hình packaging/build & khai báo entry-point CLI
├─ security_app/                      # Package chính
│  ├─ config.py                       # Hằng số: dấu lệnh "$ ", số top fail, thư mục logs, timeout mặc định
│  ├─ models.py                       # Dataclass Rule (chuẩn hoá sau parse) và CmdResult (kết quả lệnh)
│  ├─ app/			      # Main
│  │  └─ cli.py                       # Điểm vào CLI: nhận args → parse_file → run_all_rules → print_report
│  ├─ parsers/                        # Bộ phân tích theo định dạng tệp
│  │  ├─ dispatch.py                  # parse_file(): auto-chọn parser theo loại tệp
│  │  ├─ common.py                    # detect_file_type theo phần mở rộng
│  │  ├─ csv_parser.py                # Đọc CSV (pandas), chuẩn hoá cột, map → Rule (title fallback name)
│  │  ├─ json_parser.py               # Đọc JSON (stig.findings), validate trường bắt buộc → Rule
│  │  └─ xml_parser.py                # Đọc XCCDF (namespace), validate thiếu trường → Rule
│  ├─ core/                           # Lõi thực thi
│  │  ├─ runner.py                    # Orchestrator: trích lệnh cho từng Rule, chạy, log, gom số liệu
│  │  ├─ command.py                   # Chạy 1 lệnh shell, thu stdout/stderr/RC/thời gian → CmdResult
│  │  ├─ command_extractor.py         # Tách các lệnh từ “check” theo marker, hỗ trợ nối dòng "\"
│  │  └─ logger.py                    # Ghi log theo rule + summary.jsonl/csv + meta cho mỗi phiên
│  ├─ reporting/                      # Xuất kết quả kiểm ra tra các định dạng khác nhau
│  │  └─ terminal.py                  # Tính thống kê & in báo cáo: totals, by-severity, top failing rules
│  ├─ utils/                          # Chứa các hàm tiện ích
│  │   └─ text.py                     # _term_width, _safe_name, _bar, _table (in bảng/bar trên terminal)
│  └─ test/                           # Chứa các hàm test, kiểm tra giai đoạn, đo đạc thời gian, hiệu năng, tài nguyên sử dụng
└─security_app.egg-info/             # Metadata do setuptools sinh ra (PKG-INFO, SOURCES, entry_points…)
